#!/usr/bin/env python3
"""Issue #41 probe: step time of two concurrent streams by routing overlap.

decodebench.py sends one wave of identical prompts as separate HTTP requests; the
second request usually lands one engine step later, so the two streams decode
different positions. This probe controls that:

  c1          one pre-tokenized prompt (A, or B with --prompt B) via /v1/completions
  pair-same   ONE /v1/completions request with prompt [A, A]: same step, same prompt
  pair-diff   ONE /v1/completions request with prompt [A, B]: same step, other prompt
  stagger     two separate requests with prompt A, the second --stagger s later

--kind code: A is decodebench's code prompt, B a second function list.
--kind prose: A is decodebench's prose prompt, B a second explanation.
Per stream it keeps every content-chunk arrival time and the text. Step time =
median interval between consecutive chunks inside the window where every stream of
the wave runs. first_offset_ms = spread of the streams' first chunks (same prefill
step -> ~0). first_diff_char = first character where the texts of streams 0 and 1
differ (-1: identical). Metric deltas (prompt/generation tokens, finished requests)
flag outside traffic. Standard library only (safe on the head while the server runs).
"""

import argparse
import json
import re
import statistics
import threading
import time
import urllib.request

PROMPTS = {
    "code": (
        "Output only Python source code. No comments, no docstrings, no markdown "
        "fences. Write functions clamp_00 through clamp_49. Each function is exactly:\n"
        "def clamp_NN(x, lo=0, hi=1):\n"
        "    if x < lo:\n"
        "        return lo\n"
        "    if x > hi:\n"
        "        return hi\n"
        "    return x\n"
        "Change only the function name suffix (00, 01, … 49). "
        "One blank line between functions. No other text.",
        "Output only Python source code. No comments, no docstrings, no markdown "
        "fences. Write methods wrap_index_10 through wrap_index_59 of no class, as "
        "plain functions. Each function is exactly:\n"
        "def wrap_index_NN(value, size):\n"
        "    while value >= size:\n"
        "        value -= size\n"
        "    while value < 0:\n"
        "        value += size\n"
        "    return value\n"
        "Change only the function name suffix (10, 11, … 59). "
        "One blank line between functions. No other text.",
    ),
    "prose": (
        "Write a detailed step-by-step explanation of how a hash map works, "
        "including collision handling, resizing, and time complexity. Be thorough.",
        "Write a detailed step-by-step explanation of how a B-tree index works in a "
        "database, including node splits, merges, and lookup cost. Be thorough.",
    ),
}
METRICS = (
    "vllm:prompt_tokens_total",
    "vllm:generation_tokens_total",
    "vllm:request_success_total",
    "vllm:num_requests_running",
)


def post(base, path, body, timeout=60):
    req = urllib.request.Request(
        base + path,
        data=json.dumps(body).encode(),
        headers={"Content-Type": "application/json"},
    )
    return urllib.request.urlopen(req, timeout=timeout)


def metrics(base):
    text = urllib.request.urlopen(base + "/metrics", timeout=10).read().decode()
    out = dict.fromkeys(METRICS, 0.0)
    for line in text.splitlines():
        if line.startswith("#"):
            continue
        m = re.match(r"^([a-zA-Z_:]+)(\{[^}]*\})? ([0-9.eE+-]+)$", line)
        if m and m.group(1) in out:
            out[m.group(1)] += float(m.group(3))
    return out


def tokenize(base, model, prompt):
    body = {
        "model": model,
        "messages": [{"role": "user", "content": prompt}],
        "add_generation_prompt": True,
        "chat_template_kwargs": {"thinking": False},
    }
    with post(base, "/tokenize", body) as r:
        return json.load(r)["tokens"]


def completions(base, model, prompts, gen, streams, lock, key_offset=0):
    body = {
        "model": model,
        "prompt": prompts,
        "max_tokens": gen,
        "temperature": 0,
        "top_p": 1,
        "ignore_eos": True,
        "stream": True,
    }
    local = {}
    with post(base, "/v1/completions", body, timeout=600) as resp:
        for raw in resp:
            line = raw.decode("utf-8", "replace").strip()
            if not line.startswith("data:") or line == "data: [DONE]":
                continue
            try:
                obj = json.loads(line[5:])
            except json.JSONDecodeError:
                continue
            now = time.time()
            for ch in obj.get("choices") or []:
                if ch.get("text"):
                    local.setdefault(ch.get("index", 0), []).append((now, ch["text"]))
    with lock:
        for k, v in local.items():
            streams[k + key_offset] = v


def first_diff(a, b):
    for i, (x, y) in enumerate(zip(a, b)):
        if x != y:
            return i
    return -1 if len(a) == len(b) else min(len(a), len(b))


def summarize(streams):
    order = [v for _, v in sorted(streams.items())]
    times = [[t for t, _ in s] for s in order]
    texts = ["".join(x for _, x in s) for s in order]
    lo = max(t[0] for t in times)
    hi = min(t[-1] for t in times)
    per = []
    for t, text in zip(times, texts):
        iv = [b - a for a, b in zip(t, t[1:]) if a >= lo and b <= hi]
        per.append(
            {
                "chunks": len(t),
                "first": round(t[0], 4),
                "last": round(t[-1], 4),
                "win_intervals": len(iv),
                "step_med_ms": round(1e3 * statistics.median(iv), 2) if iv else None,
                "step_mean_ms": round(1e3 * statistics.mean(iv), 2) if iv else None,
                "text_len": len(text),
            }
        )
    rec = {
        "streams": per,
        "first_offset_ms": round(1e3 * (lo - min(t[0] for t in times)), 1),
        "window_s": round(hi - lo, 3),
    }
    if len(texts) > 1:
        rec["first_diff_char"] = first_diff(texts[0], texts[1])
    return rec


def wave(a, base, model, toks, mode):
    streams, lock = {}, threading.Lock()
    threads = []
    if mode == "c1":
        prompts = [[toks[a.prompt]]]
    elif mode == "pair-same":
        prompts = [[toks["A"], toks["A"]]]
    elif mode == "pair-diff":
        prompts = [[toks["A"], toks["B"]]]
    elif mode == "stagger":
        prompts = [[toks["A"]], [toks["A"]]]
    else:
        raise SystemExit(f"unknown mode {mode}")
    for i, p in enumerate(prompts):
        threads.append(
            threading.Thread(
                target=completions, args=(base, model, p, a.gen, streams, lock, i)
            )
        )
    before = metrics(base)
    t0 = time.time()
    for i, t in enumerate(threads):
        if i:
            time.sleep(a.stagger)
        t.start()
    for t in threads:
        t.join()
    t1 = time.time()
    after = metrics(base)
    rec = {
        "tag": a.tag,
        "kind": a.kind,
        "mode": mode,
        "prompt": a.prompt if mode == "c1" else None,
        "ts": time.strftime("%Y-%m-%dT%H:%M:%S", time.localtime(t0)),
        "wall_s": round(t1 - t0, 3),
        "running_before": before["vllm:num_requests_running"],
        "delta_prompt_tokens": after["vllm:prompt_tokens_total"]
        - before["vllm:prompt_tokens_total"],
        "delta_generation_tokens": after["vllm:generation_tokens_total"]
        - before["vllm:generation_tokens_total"],
        "delta_finished": after["vllm:request_success_total"]
        - before["vllm:request_success_total"],
    }
    if streams:
        rec.update(summarize(streams))
    return rec


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--base", default="http://127.0.0.1:8888")
    ap.add_argument("--tag", required=True)
    ap.add_argument("--kind", default="code", choices=tuple(PROMPTS))
    ap.add_argument("--modes", default="c1,pair-same,pair-diff")
    ap.add_argument("--reps", type=int, default=1)
    ap.add_argument("--prompt", default="A", choices=("A", "B"))
    ap.add_argument("--gen", type=int, default=256)
    ap.add_argument("--gap", type=float, default=3.0)
    ap.add_argument("--stagger", type=float, default=1.0)
    ap.add_argument("--wait-idle", type=float, default=60.0)
    ap.add_argument("--warmup", action="store_true")
    ap.add_argument("--out", default="decode_i41.jsonl")
    a = ap.parse_args()
    base = a.base.rstrip("/")
    with urllib.request.urlopen(base + "/v1/models", timeout=30) as r:
        model = json.load(r)["data"][0]["id"]
    pa, pb = PROMPTS[a.kind]
    toks = {"A": tokenize(base, model, pa), "B": tokenize(base, model, pb)}
    print(f"{a.kind} prompt tokens A={len(toks['A'])} B={len(toks['B'])}", flush=True)
    if a.warmup:
        completions(base, model, [toks["A"], toks["B"]], 32, {}, threading.Lock())
        time.sleep(a.gap)
    for _ in range(a.reps):
        for mode in a.modes.split(","):
            deadline = time.time() + a.wait_idle
            while metrics(base)["vllm:num_requests_running"] > 0:
                if time.time() > deadline:
                    break
                time.sleep(1.0)
            rec = wave(a, base, model, toks, mode)
            with open(a.out, "a") as f:
                f.write(json.dumps(rec) + "\n")
            s = rec.get("streams") or []
            print(
                f"{rec['ts']} {a.kind:5s} {mode:9s} "
                f"steps_med={[x['step_med_ms'] for x in s]} "
                f"first_off={rec.get('first_offset_ms')} ms "
                f"diff={rec.get('first_diff_char')} chunks={[x['chunks'] for x in s]} "
                f"run0={rec['running_before']:.0f} dP={rec['delta_prompt_tokens']:.0f} "
                f"dG={rec['delta_generation_tokens']:.0f} dF={rec['delta_finished']:.0f}",
                flush=True,
            )
            time.sleep(a.gap)


if __name__ == "__main__":
    main()
